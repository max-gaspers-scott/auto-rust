// crates/goose-mcp/src/comment_fixer.rs

use async_trait::async_trait;
use serde::Deserialize;
use serde_json::Value;
use std::fs;

use crate::extension::{Extension, ExtensionResult};
use crate::tool::{Tool, ToolResult};

// 1. Define the parameters the LLM must provide
#[derive(Debug, Deserialize)]
pub struct FixCommentArgs {
    pub file_path: String,
    pub target_word: String,
    pub replacement: String,
}

// 2. Define your Extension struct
pub struct PythonCommentFixerExtension;

impl PythonCommentFixerExtension {
    pub fn new() -> Self {
        Self
    }
}

// 3. Your core logic function
fn fix_python_comments(content: &str, target_word: &str, replacement: &str) -> String {
    content
        .lines()
        .map(|line| {
            if let Some(comment_index) = line.find('#') {
                let (code_part, comment_part) = line.split_at(comment_index);
                let fixed_comment = comment_part.replace(target_word, replacement);
                format!("{}{}", code_part, fixed_comment)
            } else {
                line.to_string()
            }
        })
        .collect::<Vec<_>>()
        .join("\n")
}

#[async_trait]
impl Extension for PythonCommentFixerExtension {
    fn name(&self) -> &str {
        "python_comment_fixer"
    }

    fn description(&self) -> &str {
        "Provides precise tools for safely modifying Python comments without touching source code."
    }

    // 4. Expose the tool schema to Goose's LLM prompt
    fn tools(&self) -> Vec<Tool> {
        vec![Tool::new(
            "fix_python_comment_spelling",
            "Replaces misspelled words inside Python comments (#) without altering executable python code.",
            serde_json::json!({
                "type": "object",
                "properties": {
                    "file_path": {
                        "type": "string",
                        "description": "Relative or absolute path to the .py file"
                    },
                    "target_word": {
                        "type": "string",
                        "description": "The misspelled word to search for inside comments"
                    },
                    "replacement": {
                        "type": "string",
                        "description": "The correct replacement word"
                    }
                },
                "required": ["file_path", "target_word", "replacement"]
            }),
        )]
    }

    // 5. Handle the LLM tool execution request
    async fn call(&self, tool_name: &str, arguments: Value) -> ExtensionResult<ToolResult> {
        match tool_name {
            "fix_python_comment_spelling" => {
                // Parse arguments provided by the LLM
                let args: FixCommentArgs = serde_json::from_value(arguments)
                    .map_err(|e| format!("Failed to parse arguments: {}", e))?;

                // Read the target file
                let content = fs::read_to_string(&args.file_path)
                    .map_err(|e| format!("Failed to read file '{}': {}", args.file_path, e))?;

                // Execute your custom Rust logic
                let updated_content = fix_python_comments(&content, &args.target_word, &args.replacement);

                // Save back to the file
                fs::write(&args.file_path, updated_content)
                    .map_err(|e| format!("Failed to write to file '{}': {}", args.file_path, e))?;

                // Return a confirmation string to Goose so it knows the task succeeded
                Ok(ToolResult::Text(format!(
                    "Successfully replaced '{}' with '{}' in comments inside {}",
                    args.target_word, args.replacement, args.file_path
                )))
            }
            _ => Err(format!("Unknown tool: {}", tool_name).into()),
        }
    }
}
